cmake_minimum_required(VERSION 3.20)
project(CelesteXR C)
if(NOT SDL_SOURCE_DIR OR NOT SDL_BUILD_DIR)
  message(FATAL_ERROR "Set SDL_SOURCE_DIR and SDL_BUILD_DIR to an XR-enabled SDL build")
endif()
add_library(CelesteXR SHARED celestexr.c)
target_include_directories(CelesteXR PRIVATE "${SDL_SOURCE_DIR}/include" "${SDL_SOURCE_DIR}/src/video/khronos")
target_link_libraries(CelesteXR PRIVATE "${SDL_BUILD_DIR}/Release/SDL3.lib")
add_executable(TouchVirtualGamepadTest "${CMAKE_CURRENT_SOURCE_DIR}/../../Tests/TouchVirtualGamepad.c")
target_include_directories(TouchVirtualGamepadTest PRIVATE "${SDL_SOURCE_DIR}/include")
target_link_libraries(TouchVirtualGamepadTest PRIVATE "${SDL_BUILD_DIR}/Release/SDL3.lib")
add_executable(VrPanelPoseTest "${CMAKE_CURRENT_SOURCE_DIR}/../../Tests/VrPanelPoseTest.c")
